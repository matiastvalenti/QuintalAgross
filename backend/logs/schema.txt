CREATE TABLE delivery_note_lines (
	id VARCHAR NOT NULL, 
	delivery_note_id VARCHAR NOT NULL, 
	product_id VARCHAR, 
	description VARCHAR, 
	qty NUMERIC(12, 2) NOT NULL, 
	unit_price NUMERIC(14, 2), 
	discount_pct NUMERIC(5, 2), 
	vat_rate NUMERIC(5, 4), 
	net_amount NUMERIC(14, 2), 
	vat_amount NUMERIC(14, 2), 
	total_amount NUMERIC(14, 2), 
	qty_invoiced NUMERIC(12, 2), 
	source_sales_line_id VARCHAR, 
	source_purchase_line_id VARCHAR, 
	line_order INTEGER, unit_cost VARCHAR, total_cost VARCHAR, 
	PRIMARY KEY (id), 
	FOREIGN KEY(delivery_note_id) REFERENCES delivery_notes (id), 
	FOREIGN KEY(product_id) REFERENCES products (id), 
	FOREIGN KEY(source_sales_line_id) REFERENCES sales_order_lines (id), 
	FOREIGN KEY(source_purchase_line_id) REFERENCES purchase_order_lines (id)
)
